cmake_minimum_required(VERSION 3.23)

project(opencv_project LANGUAGES CXX)

set(OpenCV_STATIC OFF)
find_package(OpenCV REQUIRED)

add_executable(opencv_project main.cpp)
target_link_libraries(opencv_project PRIVATE ${OpenCV_LIBS})

add_custom_command(TARGET opencv_project POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
    $<TARGET_RUNTIME_DLLS:opencv_project>
    $<TARGET_FILE_DIR:opencv_project>
    COMMAND_EXPAND_LISTS
)
